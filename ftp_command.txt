//git ftp文檔
https://github.com/git-ftp/git-ftp/blob/master/INSTALL.md

//安裝git ftp
//管理員開啟 Git Bash 輸入已下指令
curl https://raw.githubusercontent.com/git-ftp/git-ftp/master/git-ftp > /bin/git-ftp
chmod 755 /bin/git-ftp

//推正式機
git ftp init -u admin@mangotoolweb.com -p '[zUAjeYbXu)%' ftp://ftp.mangotoolweb.com/public_html/
git ftp push -u admin@mangotoolweb.com -p '[zUAjeYbXu)%' ftp://ftp.mangotoolweb.com/public_html/
//設定主機
git config git-ftp.url ftp://ftp.mangotoolweb.com/public_html/
git config git-ftp.user admin@mangotoolweb.com
git config git-ftp.password '[zUAjeYbXu)%'
//設定本機路徑
git config git-ftp.syncroot C:\xampp\htdocs\


//初始化
git ftp init
//推測試機
git ftp push
