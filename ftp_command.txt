//git ftp文檔
https://github.com/git-ftp/git-ftp/blob/master/INSTALL.md

//安裝git ftp
//管理員開啟 Git Bash 輸入已下指令
curl https://raw.githubusercontent.com/git-ftp/git-ftp/master/git-ftp > /bin/git-ftp
chmod 755 /bin/git-ftp

//推正式機
git ftp push -u dogonthemoon -p 'mTxmT3YYymSmPbdz'  ftp://34.122.138.173:21
//設定主機
git config git-ftp.url ftp://34.122.138.173:21
git config git-ftp.user dogonthemoon
git config git-ftp.password 'mTxmT3YYymSmPbdz'
//設定本機路徑
git config git-ftp.syncroot C:\xampp\htdocs\


//初始化
git ftp init
//推測試機
git ftp push
